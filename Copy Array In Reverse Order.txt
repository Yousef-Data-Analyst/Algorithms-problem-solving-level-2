#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

int readPositives(const string &message)
{
    int number;
    do
    {
        cout << message;

        cin >> number;
    } while (number <= 0);

    return number;
}
int randomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void printArray(int arr[], int size)
{

    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
}

void fillArrayRandomly(int *&arr, int size)
{

    arr = new int[size];
    for (int i = 0; i < size; i++)
    {
        arr[i] = randomNumber(1, 100);
    }
}

void copyArrayReversed(int arr[], int *&copy, int size)
{
    copy = new int[size];
    int counter = 0;
    for (int i = size - 1; i >= 0; i--)
    {
        copy[counter++] = arr[i];
    }
}

int main()
{
    system("cls");

    srand((unsigned)time(NULL));

    int *arr, size;

    size = readPositives("Enter the array size: ");
    cout << "\n*****************************************\n\n";

    fillArrayRandomly(arr, size);

    cout << endl
         << "Original Array: ";
    printArray(arr, size);

    cout << endl;

    int *copy;

    copyArrayReversed(arr, copy, size);

    cout << endl
         << "Reversed Array: ";
    printArray(copy, size);

    delete[] arr;
    delete[] copy;
}
