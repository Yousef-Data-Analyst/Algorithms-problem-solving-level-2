#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

struct Table
{
    string title;
    int multiplicationNumber;
    int colWidth;
};

Table readTableData()
{
    Table table;

    cout << "Enter the multiplication number: ";
    cin >> table.multiplicationNumber;
    int minLength = to_string(table.multiplicationNumber * table.multiplicationNumber).length();

    cout << "Enter the columns width > " << minLength << ": ";
    cin >> table.colWidth;
    if (table.colWidth < minLength)
        table.colWidth = minLength + 1;

    cin.ignore();

    int maxLength = table.colWidth * table.multiplicationNumber;
    do
    {
        cout << "Enter the Table title (length <= " << maxLength << " Characters) : ";
        getline(cin, table.title);
    } while (table.title.length() > maxLength);
    return table;
}

void printTitle(const string &title, int width, int number)
{

    cout << endl
         << endl
         << setw(title.length() + (width * number - title.length()) / 2) << title << endl
         << endl;
}

void printHeaderNumbersFor(int number, int width)
{
    cout << setw(width - 1) << left << "";
    for (int i = 1; i <= number; i++)
    {
        cout << setw(width) << left << i;
    }
    cout << endl;
    for (int i = 1; i <= number; i++)
    {
        cout << setw(width) << setfill('-') << "";
    }
    cout << "--" << setfill(' ') << endl;
}

void printContent(int number, int width)
{

    for (int i = 1; i <= number; i++)
    {
        cout << setw(width / 2) << left << i << setw(width - (width / 2) - 1) << "|";
        for (int j = 1; j <= number; j++)
        {
            cout << setw(width) << i * j;
        }
        cout << endl;
    }
}

void printMultiplicationTable()
{
    system("cls");
    Table table = readTableData();
    system("cls");
    printTitle(table.title, table.colWidth, table.multiplicationNumber);
    printHeaderNumbersFor(table.multiplicationNumber, table.colWidth);
    printContent(table.multiplicationNumber, table.colWidth);
}

int main()
{
    printMultiplicationTable();
}
