#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

int readPositives(const string &message)
{
    int number;
    do
    {
        cout << message;

        cin >> number;
    } while (number <= 0);

    return number;
}
int randomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void printArray(int arr[], int size)
{

    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
}

void fillArrayFrom1_to_N(int *&arr, int n)
{

    arr = new int[n];
    for (int i = 0; i < n; i++)
    {
        arr[i] = i + 1;
    }
}

void swap(int &a, int &b)
{
    int tmp = a;
    a = b;
    b = tmp;
}

void shuffle(int arr[], int size)
{
    for (int i = 0; i < size; i++)
    {
        swap(arr[i], arr[randomNumber(0, size - 1)]);
    }
}

int main()
{
    system("cls");

    srand((unsigned)time(NULL));

    int *arr, size;

    size = readPositives("Enter the array size: ");
    cout << "\n*****************************************\n\n";

    fillArrayFrom1_to_N(arr, size);

    cout << "\nOriginal Array: ";
    printArray(arr, size);

    cout << endl;

    shuffle(arr, size);

    cout << "\nShuffled Array: ";
    printArray(arr, size);
}
