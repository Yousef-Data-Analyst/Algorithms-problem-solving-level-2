#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

void printHeader()
{
    cout << endl
         << "\t\t\tMultiplication Table from 1 to 10 :)" << endl;
    for (int i = 1; i <= 10; i++)
    {
        cout << "\t" << i;
    }
    cout << "\n------------------------------------------------------------------------------------\n";
}

string separator(int num)
{
    return (num > 9) ? "  |\t" : "   |\t";
}
void printContent()
{
    for (int i = 1; i <= 10; i++)
    {
        cout << i << separator(i);
        for (int j = 1; j <= 10; j++)
        {
            cout << i * j << "\t";
        }
        cout << endl;
    }
}

int main()
{
    system("cls");
    printHeader();
    printContent();
    
}
