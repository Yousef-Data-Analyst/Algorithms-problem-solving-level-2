#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

float readNumber(const string &message, bool positive = true)
{
    float number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

float myAbs(float number)
{
    return (number < 0) ? number * -1 : number;
}

int main()
{
    system("cls");

    float number = readNumber("Enter a number: ", false);

    cout << endl
         << "*********************************" << endl
         << endl;

    cout << "My abs result: " << myAbs(number) << endl;
    cout << "C++ abs result: " << abs(number) << endl;
}
