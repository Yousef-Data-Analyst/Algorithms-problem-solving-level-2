#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

int readNonNegative(const string &message)
{
    int number;
    do
    {
        cout << message;
        cin >> number;

    } while (number < 0);
    return number;
}

void printNumberPattern(int num)
{
    for (int i = 1; i <= num; i++)
    {
        for (int j = 1; j <= i; j++)
            cout << i << " ";
        cout << endl;
    }
}

int main()
{
    system("cls");
    printNumberPattern(readNonNegative("Enter a number to print an inverted pattern :) ? "));
}
