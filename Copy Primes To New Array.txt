#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

int readInteger(const string &message, bool positive = true)
{
    int number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

void addArrayElement(int element, int arr[], int &position)
{
    arr[position++] = element;
}

int randomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void fillArrayRandomly(int arr[], int &size)
{
    for (int i = 0; i < size; i++)
    {
        arr[i] = randomNumber(1, 100);
    }
}

void printArray(int arr[], int size)
{
    cout << "[";
    for (int i = 0; i < size; i++)
    {
        cout << arr[i];
        if (i == size - 1)
            continue;
        cout << ", ";
    }
    cout << "]";
}

bool isPrime(int number)
{
    if (number == 1)
        return false;
    if (number == 2)
        return true;
    if (number % 2 == 0)
        return false;
    for (int i = 3; i <= sqrt(number); i += 2)
    {
        if (number % i == 0)
            return false;
    }
    return true;
}

void copyPrimeElements(int arr[], int copy[], int arrSize, int &copySize)
{
    int counter = 0;
    for (int i = 0; i < arrSize; i++)
    {
        if (isPrime(arr[i]))
            addArrayElement(arr[i], copy, counter);
    }
    copySize = counter;
}

int main()
{
    system("cls");

    srand((unsigned)time(NULL));

    int arr[100], size;
    size = readInteger("Enter the array size: ");

    fillArrayRandomly(arr, size);

    cout << endl
         << "Array Elements: ";
    printArray(arr, size);

    cout << endl;

    int copy[100], copySize;

    copyPrimeElements(arr, copy, size, copySize);

    cout << endl
         << "Copy Elemetns: ";
    printArray(copy, copySize);
    cout << endl;
}
