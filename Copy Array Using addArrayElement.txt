#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

int readInteger(const string &message, bool positive = true)
{
    int number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

void addArrayElement(int element, int arr[], int &position)
{
    arr[position++] = element;
}

int randomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void fillArrayRandomly(int arr[], int &size)
{
    for (int i = 0; i < size; i++)
    {
        arr[i] = randomNumber(1, 100);
    }
}

void printArray(int arr[], int size)
{
    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
}

void copyArrayElements(int arr[], int copy[], int size)
{
    int counter = 0;
    for (int i = 0; i < size; i++)
    {
        addArrayElement(arr[counter], copy, counter);
    }
}
int main()
{
    system("cls");

    srand((unsigned)time(NULL));

    int arr[100], size;
    size = readInteger("Enter the array size: ");

    fillArrayRandomly(arr, size);

    cout << endl
         << "Array Elements: ";
    printArray(arr, size);

    cout << endl;

    int copy[100];

    copyArrayElements(arr, copy, size);

    cout << endl
         << "Copy Elemetns: ";
    printArray(copy, size);
    cout << endl;
}
