#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

long long readNonNegative(const string &message)
{
    long long number;
    cout << message;
    cin >> number;
    return number;
}

long long reverseDigits(long long number)
{
    long long reversedNumber = 0;
    while (number > 0)
    {
        reversedNumber = reversedNumber * 10 + number % 10;
        number /= 10;
    }
    return reversedNumber;
}

void printDigitsInOrderFor(long long number)
{
    int length = to_string(number).length();
    long long a;
    do
    {
        a = number;
        a /= pow(10, --length);
        cout << a % 10 << endl;
    } while (length > 0);
}

void printDigitsInOrderReversedApproach(long long number)
{
    do
    {
        cout << number % 10 << endl;
        number /= 10;
    } while (number > 0);
}

int main()
{
    system("cls");
    printDigitsInOrderFor(readNonNegative("Enter a number to print: "));
    cout << endl << "********************************"<< endl<<endl;
    printDigitsInOrderReversedApproach(reverseDigits(readNonNegative("Enter a number to print: ")));
}
