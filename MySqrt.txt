#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

float readNumber(const string &message, bool positive = true)
{
    float number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

float getFractionPart(float number)
{
    return abs(number - int(number));
}

float mySqrt(float number)
{
    return pow(number, .5);
}

int main()
{
    system("cls");
    float number = readNumber("Enter a Number: ", false);
    cout << "MySqrt result: " << mySqrt(number) << endl;
    cout << "C++ sqrt result: " << sqrt(number) << endl;
}
