#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

float readNumber(const string &message, bool positive = true)
{
    float number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

int getFractionNumber(float number)
{
    return (number > 0) ? int(number * 10) % 10 : int(-number * 10) % 10;
}

float myRound(float number)
{
    int fraction = getFractionNumber(number);
    int intPart = int(number);
    if (fraction >= 5)
        // roundup
        return (intPart > 0) ? ++intPart : --intPart;
    else
        // roundown
        return intPart;
}

int main()
{
    system("cls");
    float number = readNumber("Enter a Number to round: ", false);
    cout << "MyRound result: " << myRound(number) << endl;
    cout << "C++ Round result: " << round(number) << endl;
}
