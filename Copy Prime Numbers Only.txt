#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

int readPositives(const string &message)
{
    int number;
    do
    {
        cout << message;

        cin >> number;
    } while (number <= 0);

    return number;
}
int randomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void fillArrayRandomly(int *&arr, int size)
{
    arr = new int[size];
    for (int i = 0; i < size; i++)
    {
        arr[i] = randomNumber(1, 100);
    }
}

void printArray(int arr[], int size)
{

    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
}

bool isPrime(int number)
{
    if (number == 1)
        return false;
    if (number == 2)
        return true;
    if (number % 2 == 0)
        return false;
    for (int i = 3; i <= sqrt(number); i += 2)
    {
        if (number % i == 0)
            return false;
    }
    return true;
}

int copyPrimes(int arr[], int *&arrayCopy, int size)
{
    arrayCopy = new int[size];
    int counter = 0;
    for (int i = 0; i < size; i++)
    {
        if (isPrime(arr[i]))
            arrayCopy[counter++] = arr[i];
    }
    return counter;
}

int main()
{
    system("cls");

    srand((unsigned)time(NULL));

    int *arr;

    int size = readPositives("Enter the array size: ");

    cout << endl
         << "***********************************" << endl;

    fillArrayRandomly(arr, size);

    cout << "\nArray Elements: ";
    printArray(arr, size);

    int *copy;

    cout << endl;
    cout << "Copied Primes: ";
    printArray(copy, copyPrimes(arr, copy, size));

    delete[] arr;
    delete[] copy;
}
