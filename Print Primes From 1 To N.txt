#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

bool isPrime(int number)
{
    if (number == 1)
        return false;
    if (number == 2)
        return true;
    if (number % 2 == 0)
        return false;

    for (int i = 3; i <= pow(number, .5); i++)
    {
        if (number % i == 0)
            return false;
    }
    return true;
}

void printPrimeNumbersFor(int number)
{
    for (int i = 1; i <= number; i++)
    {
        if (isPrime(i))
        {
            cout << i << endl;
        }
    }
}

int readNonNegativeNumber(const string &message)
{
    cout << message;
    int number;
    cin >> number;
    return abs(number);
}
int main()
{
    system("cls");
    printPrimeNumbersFor(readNonNegativeNumber("Enter a number to get the primes untill it: "));
}
