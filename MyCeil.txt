#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

float readNumber(const string &message, bool positive = true)
{
    float number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

float getFractionPart(float number)
{
    return abs(number - int(number));
}

float myCeil(float number)
{
    if (getFractionPart(number) > 0)
    {
        if (number > 0)
            return int(number) + 1;
        else
            return int(number);
    }
    else
        return int(number);

    // if (number == int(number) || number < 0)
    //     return int(number);
    // else
    //     return int(number) + 1;
}

int main()
{
    system("cls");
    float number = readNumber("Enter a Number: ", false);
    cout << "MyCeil result: " << myCeil(number) << endl;
    cout << "C++ ceil result: " << ceil(number) << endl;
}
