int readInteger(const string &message, bool positive = true)
{
    int number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

void addArrayElement(int element, int arr[], int &position)
{
    arr[position++] = element;
}

int randomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void fillArray(int arr[], int &size)
{
    size = 6;
    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;
    arr[3] = 30;
    arr[4] = 20;
    arr[5] = 10;
}

void printArray(int arr[], int size)
{
    cout << "[";
    for (int i = 0; i < size; i++)
    {
        cout << arr[i];
        if (i == size - 1)
            continue;
        cout << ", ";
    }
    cout << "]";
}

bool isPalindrome(int arr[], int size)
{
    for (int i = 0; i < size / 2; i++)
    {
        if (arr[i] != arr[size - 1 - i])
            return false;
    }
    return true;
}

void printResult(bool ispalindrome)
{
    if (ispalindrome)
        cout << endl
             << "Yes Array is Palindrome :)" << endl;
    else
        cout << endl
             << "No Arrya is not Palindrome :(" << endl;
}
