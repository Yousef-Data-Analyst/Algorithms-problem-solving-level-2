#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

string readPassword(const string &message)
{
    string password;
    cout << message;
    cin >> password;
    return password;
}

void getPassword(string password)
{
    char letter = 'A';
    string guess = "";
    int trial = 0;
    bool found = false;
    for (int i = 0; i < 26; i++)
    {
        if (found)
            break;
        for (int j = 0; j < 26; j++)
        {
            if (found)
                break;
            for (int k = 0; k < 26; k++)
            {
                guess += char(letter + i);
                guess += char(letter + j);
                guess += char(letter + k);
                cout << "Trial[" << ++trial << "]: " << guess << endl;
                if (guess == password)
                {
                    found = true;
                    cout << "Password is " << guess << endl;
                    cout << "Found after " << trial << " Trials." << endl;

                    break;
                }
                guess = "";
            }
        }
    }
}

int main()
{
    system("cls");
    getPassword(readPassword("Enter a 3 english letter password to guess: "));
}
