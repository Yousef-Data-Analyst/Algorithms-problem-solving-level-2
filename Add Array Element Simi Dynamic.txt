#include <iostream>
#include <cstdlib>
#include <cmath>
#include <string>

using namespace std;

int readInteger(const string &message, bool positive = true)
{
    int number;

    do
    {
        cout << message;
        cin >> number;

    } while (positive && number < 0);

    return number;
}

bool addMore()
{

    int number = readInteger("Do you wanna add another number (yes = 1, no = 0): ");
    return number == 1;
}

void addArrayElement(int element, int arr[], int &position)
{
    arr[position++] = element;
}

void inputUserArray(int arr[], int &size)
{
    int counter = 0;
    int number;
    do
    {
        addArrayElement(readInteger("Enter a number: "), arr, counter);
    } while (addMore() && counter < 100);
    size = counter;
}

void printArray(int arr[], int size)
{
    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
}

int main()
{
    system("cls");

    srand((unsigned)time(NULL));

    int arr[100], size;

    inputUserArray(arr, size);

    cout << "\nArray Length: " << size << endl;

    cout << endl
         << "Array Elements: " << endl;
    printArray(arr, size);
}
