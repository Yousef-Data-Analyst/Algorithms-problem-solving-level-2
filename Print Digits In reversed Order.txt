#include <iostream>
#include <cmath>
#include <string>
#include <iomanip>
using namespace std;

long long readNonNegativeNumber(const string &message)
{
    long long number = 0;
    do
    {
        cout << message;
        cin >> number;
    } while (number < 0);
    return number;
}

long long reverseDigits(long long number)
{
    long long reversedNumber = 0;
    while (number > 0)
    {
        reversedNumber = reversedNumber * 10 + number % 10;
        number /= 10;
        }
    return reversedNumber;
}

int main()
{

    system("cls");
    long long number = readNonNegativeNumber("Enter a number to reverse its digits: ");
    cout << endl
         << "************************************************" << endl
         << endl;
    cout << "Reversed Number: " << reverseDigits(number) << endl
         << endl;
}
