int readNonNegative(const string &message)

    } while (number < 0);

}

void printLettersFromAAA_To_ZZZ()
{
    char letter = 'A';
    for (int i = 0; i < 26; i++)
    {
        for (int j = 0; j < 26; j++)
        {
            for (int k = 0; k < 26; k++)
            {
                cout << char(letter + i) << " " << char(letter + j) << " " << char(letter + k) << " " << endl;
            }
        }
    }
}
