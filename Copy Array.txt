#include <iostream>
#include <cstdlib>
#include <string>

using namespace std;

int readPositives(const string &message)
{
    int number;
    do
    {
        cout << message;

        cin >> number;
    } while (number <= 0);

    return number;
}
int radomNumber(int from, int to)
{
    return rand() % (to - from + 1) + from;
}

void fillArrayRandomly(int *&arr, int size)
{
    arr = new int[size];
    for (int i = 0; i < size; i++)
    {
        arr[i] = radomNumber(1, 100);
    }
}

void printArray(int arr[], int size)
{

    for (int i = 0; i < size; i++)
    {
        cout << arr[i] << " ";
    }
}

void copyArray(int arr[], int *&arrayCopy, int size)
{
    arrayCopy = new int[size];
    for (int i = 0; i < size; i++)
    {
        arrayCopy[i] = arr[i];
    }
}

int main()
{
    system("cls");
    srand((unsigned)time(NULL));
    int *arr;
    int size = readPositives("Enter the array size: ");

    cout << endl
         << "***********************************" << endl;

    fillArrayRandomly(arr, size);

    cout << "\nArray Elements: ";
    printArray(arr, size);

    int *copy;
    copyArray(arr, copy, size);

    cout << endl;

    cout << "Copy Elements: ";
    printArray(copy, size);
    delete[] arr;
}
